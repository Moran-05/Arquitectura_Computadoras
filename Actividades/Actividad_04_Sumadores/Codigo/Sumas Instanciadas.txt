module half_adder( 
	input a, 
	input b,
	output s,
	output c
	);
	
	assign s = (a^b);
	assign c = a & b;
	
endmodule

//creacion de modulo I/O
module full_adder(
	input a,
	input b,
	input ci,
	output s,
	output co
	);
//Declaracion de wires, y registros

wire s1;
wire c1;
wire c2;

//assigns, instancias, bloques secuenciales.

half_adder lonche( .a(a),.b(b),.s(s1),.c(c1) );
half_adder torta( .a(ci),.b(s1),.s(s),.c(c2) );
assign co = c1 | c2;

endmodule

module nibble_adder (
	input [3:0] a,
	input [3:0] b,
	input ci,
	output [3:0] s,
	output co
);

wire c1;
wire c2;
wire c3;

full_adder fa0 (.a(a[0]),.b(b[0]),.ci(ci),.s(s[0]),.co(c1));
full_adder fa1 (.a(a[1]),.b(b[1]),.ci(c1),.s(s[1]),.co(c2));
full_adder fa2 (.a(a[2]),.b(b[2]),.ci(c2),.s(s[2]),.co(c3));
full_adder fa3 (.a(a[3]),.b(b[3]),.ci(c3),.s(s[3]),.co(co));

endmodule

module byte_adder(
	input [7:0] a,
	input [7:0] b,
	output [8:0] c
);

wire c1;

nibble_adderC na0(.a(a[3:0]), .b(b[3:0]), .ci(0), .co({c1, c[3:0]}));
nibble_adderC na1 (.a(a[7:4]), .b(b[7:4]), .ci(c1), .co(c[8:4]));

endmodule
