module nibble_adderC (
	input [3:0] a,
	input [3:0] b,
	input ci,
	output [4:0] co
);

assign co = a + b + ci;

endmodule

module byte_adderC(
    input [7:0] a,
	input [7:0] b,
	output [8:0] c
    );
    
    assign c = a + b;
    
endmodule

`timescale 1ns / 1ps
module TB_nibble_adder();
reg [3:0] a;
reg [3:0] b;
reg ci;
wire [4:0] c;

nibble_adderC na1(.a(a[3:0]), .b(b[3:0]), .ci(ci), .co(c[4:0]));

initial begin
	a = 4'd15;
	b = 4'd15;
	#100;
	a = 4'd8;
	b = 4'd10;
	#100;
	a = 4'd10;
	b = 4'd6;
	#100;
	a = 4'd7;
	b = 4'd7;
	#100;
	a = 4'd9;
	b = 4'd9;
	#100;
$stop;
end
endmodule

`timescale 1ns / 1ps
module TB_byte_adder ();

reg [7:0] a;
reg [7:0] b;
wire [8:0] c;

byte_adder ba1(.a(a[7:0]), .b(b[7:0]), .c(c[8:0]));

initial begin
	a = 8'd150;
	b = 8'd30;
	#100;
	a = 8'd200;
	b = 8'd10;
	#100;
	a = 8'd98;
	b = 8'd78;
	#100;
	a = 8'd35;
	b = 8'd47;
	#100;
	a = 8'd37;
	b = 8'd77;
	#100;
$stop;
end

endmodule
